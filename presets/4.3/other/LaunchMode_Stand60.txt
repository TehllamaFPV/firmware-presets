#$ TITLE: Tehllama Launch Stand Mode Enabled
#$ FIRMWARE_VERSION: 4.2
#$ FIRMWARE_VERSION: 4.3
#$ STATUS: OFFICIAL
#$ CATEGORY: OTHER
#$ KEYWORDS: Race, Launch, LaunchMode, Mode
#$ AUTHOR: Daniel Appel / Tehllama
#$ DESCRIPTION: This is a launch stand launch-mode option that enables first-takeoff only pitch-only launch behavior
#$ DISCLAIMER: In high winds, with poor idle performance, or some stands, this will result in the craft falling off the stand.
#$ WARNING: Accelerometer Integration Feature requires enabling accelerometer in the Configuration tab.
#$ DISCUSSION: https://app.slack.com/client/T221J1F6J/C0228HK2MD4/

#$ OPTION BEGIN (CHECKED): Apply to current RateProfile Only
set launch_control_mode = PITCHONLY
set launch_trigger_allow_reset = OFF
set launch_angle_limit = 60
#$ OPTION END

#$ OPTION BEGIN (CHECKED): Apply to ALL RateProfiles (Sets RateProfile #1)
rateprofile 0
set launch_control_mode = PITCHONLY
set launch_trigger_allow_reset = OFF
set launch_angle_limit = 60
rateprofile 2
set launch_control_mode = PITCHONLY
set launch_trigger_allow_reset = OFF
set launch_angle_limit = 60
rateprofile 1
set launch_control_mode = PITCHONLY
set launch_trigger_allow_reset = OFF
set launch_angle_limit = 60
rateprofile 0
#$ OPTION END
